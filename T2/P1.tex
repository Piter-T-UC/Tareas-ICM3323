\section*{Problema 1}

\begin{enunciado}{Problema 1 (2 puntos)}
    Para la viga de acero indicada en la Figura se pide:
    \begin{enumerate}[label=\alph*), itemsep=4pt]
        \item Obtener la expresión analítica de los campos de desplazamiento,
        rotación, momento y corte de acuerdo a la teoría de vigas.
        \item Obtener la solución por MEF de dichos campos utilizando al menos
        tres discretizaciones diferentes con un número creciente de elementos
        finitos de viga de dos nodos. Justificar la elección de cada una de las
        discretizaciones propuestas.
        \item Graficar, comparar y discutir las soluciones obtenidas en a) y b).
    \end{enumerate}

    \medskip
    \noindent Datos: $E = 210$ GPa, $G = 80$ GPa, $S_y = 300$ MPa, sección

\begin{desarrollo}{a) Campos analíticos de desplazamiento, rotación, momento y corte}

\textbf{1. Ecuaciones de Gobierno y Parámetros}
\begin{align*}
    E\,I(z)\,v''(z) &= M_f(z) & G\,J(z)\,\varphi'(z) &= T(z) & \text{C.B.: } v(0) &= \theta(0) = \varphi(0) = 0
\end{align*}
Propiedades geométricas ($L=1000\text{ mm}$, $I_0=2\cdot 10^6\text{ mm}^4$, $I_1=10^6\text{ mm}^4$, $z_0 = 2000\text{ mm}$):
\begin{align*}
    I(z) &= I_0 - \frac{I_1}{L}z = 1000(2000 - z)\ \text{[mm$^4$]} & J(z) &= 2I(z) = 2000(2000 - z)\ \text{[mm$^4$]}
\end{align*}

\textbf{2. Esfuerzos Internos ($P=1000\text{ N}$, $a=500\text{ mm}$, $M=2\cdot 10^6\text{ N}\cdot\text{mm}$)}
$$V(z)=\begin{cases} 1000\text{ N} & 0\le z < 500 \\ 0 & 500 < z \le 1000 \end{cases} \qquad
M_f(z)=\begin{cases} 1000(500-z)\text{ N}\cdot\text{mm} & 0\le z\le 500 \\ 0 & 500\le z\le 1000 \end{cases}$$
$$T(z) = 2\cdot 10^6\text{ N}\cdot\text{mm} \quad (0\le z\le 1000)$$

\textbf{3. Flexión: Tramo $0 \le z \le 500\text{ mm}$ ($E = 210\,000\text{ MPa}$)}
\begin{align*}
    v''(z) &= \frac{M_f(z)}{E\,I(z)} = \frac{1000(500-z)}{210\,000 \cdot 1000(2000-z)} = \frac{1}{210\,000}\left(\frac{500-z}{2000-z}\right) = \frac{1}{210\,000}\left(1 - \frac{1500}{2000-z}\right) \\[8pt]
    \theta(z) &= \int v''(z)\,dz = \frac{1}{210\,000}\Big[z + 1500\ln(2000-z)\Big] + C_1 \\
    \theta(0) &= 0 \implies C_1 = -\frac{1500\ln(2000)}{210\,000} \implies \theta(z) = \frac{1}{210\,000}\left[z + 1500\ln\!\left(1-\frac{z}{2000}\right)\right]\ \text{[rad]} \\[8pt]
    v(z) &= \int \theta(z)\,dz = \frac{1}{210\,000}\left[\frac{z^2}{2} - 1500z - 1500(2000-z)\ln\!\left(1-\frac{z}{2000}\right)\right] + C_2 \\
    v(0) &= 0 \implies C_2 = 0 \implies v(z) = \frac{1}{210\,000}\left[\frac{z^2}{2} - 1500z - 1500(2000-z)\ln\!\left(1-\frac{z}{2000}\right)\right]\ \text{[mm]}
\end{align*}
Evaluación en el límite del tramo ($z = 500\text{ mm}$):
\begin{align*}
    \theta(500) &= \frac{1}{210\,000}\Big[500 + 1500\ln(0.75)\Big] = 3.2608\times 10^{-4}\ \text{rad} \\
    v(500) &= \frac{1}{210\,000}\Big[\frac{500^2}{2} - 1500(500) - 1500(1500)\ln(0.75)\Big] = 0.10612\ \text{mm}
\end{align*}

\textbf{4. Flexión: Tramo $500 \le z \le 1000\text{ mm}$}
\begin{align*}
    v''(z) &= 0 \implies \theta(z) = \theta(500) = 3.2608\times 10^{-4}\ \text{rad} \\
    v(z) &= v(500) + \theta(500)(z - 500) = 0.10612 + 3.2608\times 10^{-4}(z - 500)\ \text{[mm]} \\
    v(1000) &= 0.10612 + 3.2608\times 10^{-4}(500) = 0.26916\ \text{mm}
\end{align*}

\textbf{5. Torsión: Tramo $0 \le z \le 1000\text{ mm}$ ($G = 80\,000\text{ MPa}$)}
\begin{align*}
    \varphi'(z) &= \frac{T(z)}{G\,J(z)} = \frac{2\cdot 10^6}{80\,000 \cdot 2000(2000-z)} = \frac{0.0125}{2000-z} \\[6pt]
    \varphi(z) &= \int \frac{0.0125}{2000-z}\,dz = -0.0125\ln(2000-z) + C_3 \\
    \varphi(0) &= 0 \implies C_3 = 0.0125\ln(2000) \implies \varphi(z) = 0.0125\ln\!\left(\frac{2000}{2000-z}\right)\ \text{[rad]} \\[6pt]
    \varphi(1000) &= 0.0125\ln\!\left(\frac{2000}{1000}\right) = 0.0125\ln(2) = 8.6643\times 10^{-3}\ \text{rad}
\end{align*}

\begin{desarrollo}{b) Solución por MEF}

\textbf{Modelo}\\
La viga se modela con elementos de viga 3D de dos nodos (6 GDL por nodo:
$u_x,u_y,u_z,\theta_x,\theta_y,\theta_z$), en este caso se decidio crear solo 1 sistema de funciones en python para resolver, en este caso se utiliza la matriz para el caso de inercia linealmente variable. Para este problema es necesario usar minimo 2 elementos cada uno de $L/2$ por la posicion de las fuerzas y momentos.
\begin{itemize}[itemsep=2pt]
    \item \textbf{Flexión:} funciones de forma cúbicas con la
    integral $\int_0^{l} B^T E\,I(z)\,B\,dz$ calculada en forma exacta para
    $I$ lineal:
    $$\mathbf{k}_f=\frac{E}{l^3}\begin{bmatrix}
    6a & 2lb & -6a & 2lc\\
    2lb & l^2(3I_i+I_j) & -2lb & l^2a\\
    -6a & -2lb & 6a & -2lc\\
    2lc & l^2a & -2lc & l^2(I_i+3I_j)
    \end{bmatrix},\quad a=I_i+I_j,\ b=2I_i+I_j,\ c=I_i+2I_j$$
    (con $I_i=I_j=I$ se recupera la matriz clásica $12,\,6l,\,4l^2,\,2l^2$).
    \item \textbf{Torsión:} funciones de forma lineales, lo que da
    $\mathbf{k}_t=\frac{G\bar J}{l}\begin{bmatrix}1&-1\\-1&1\end{bmatrix}$ con
    $\bar J=(J_i+J_j)/2$ y $J=2I$.
\end{itemize}
El nodo 1 ($z=0$) tiene sus 6 GDL restringidos al ser un empotramiento

\textbf{Sistema reducido para $n=2$}\\
Con dos elementos de $l=0.5$ m hay 3 nodos y 18 GDL, de los cuales quedan 12
libres (nodos 2 y 3). Como la viga es recta y las cargas solo actúan en el
plano $x$--$z$ y en torsión, los GDL $u_x,u_y,\theta_z$ quedan desacoplados
y sin carga, por lo que valen cero. Basta con resolver el sistema de los GDL
$u_z,\theta_x,\theta_y$ (unidades SI: N, m, rad):
\[
10^{5}
\begin{bmatrix}
604.8 & 0 & 16.8 & -252 & 0 & -58.8 \\
0 & 9.6 & 0 & 0 & -4 & 0 \\
16.8 & 0 & 50.4 & 67.2 & 0 & 10.5 \\
-252 & 0 & 67.2 & 252 & 0 & 58.8 \\
0 & -4 & 0 & 0 & 4 & 0 \\
-58.8 & 0 & 10.5 & 58.8 & 0 & 18.9
\end{bmatrix}
\begin{bmatrix}
u_{z2} \\ \theta_{x2} \\ \theta_{y2} \\ u_{z3} \\ \theta_{x3} \\ \theta_{y3}
\end{bmatrix}
=
10^{3}
\begin{bmatrix}
-1 \\ 0 \\ 0 \\ 0 \\ 2 \\ 0
\end{bmatrix}
\]
Se verifica, por ejemplo, la torsión: $K_{\theta_{x2}\theta_{x2}}=
G(\bar J_1+\bar J_2)/l=80\times10^9\,(3.5+2.5)\times10^{-6}/0.5=9.6\times10^5$
N$\cdot$m/rad. Además, flexión y torsión no se acoplan entre sí (los ceros
de la matriz), igual que en la solución analítica.

\textbf{Discretizaciones}\\
Se usan $n=2,\,4,\,8$ y $16$ elementos de igual largo:
\begin{itemize}[itemsep=2pt]
    \item \textbf{$n=2$:} es la malla más gruesa posible, porque la carga $P$
    debe caer sobre un nodo (en $z=L/2$). Sirve para ver cuánto error comete
    el MEF con el mínimo de elementos.
    \item \textbf{$n=4$ y $n=8$:} al duplicar $n$ el largo de elemento $l$ se
    reduce a la mitad, lo que permite medir la tasa de convergencia
    $e\propto l^{p}$. $n$ siempre es par para mantener un nodo bajo $P$.
    \item \textbf{$n=16$:} malla fina para verificar que la solución ya
    convergió. El MEF no es exacto aquí porque, con $I(z)$ variable, las
    cúbicas de Hermite (y las lineales en torsión) no son la solución
    exacta de la ecuación diferencial dentro de cada elemento.
\end{itemize}

\textbf{Reconstrucción de los campos}\\
Los campos a lo largo de la viga se obtienen de los desplazamientos nodales:
$v=-u_z$ se interpola con las cúbicas de Hermite a partir de
$(v,\theta)$ de cada nodo (con $\theta=\theta_y=dv/dz$),
$\theta(z)=v'(z)$ es la derivada de esa interpolación y $\varphi=\theta_x$
se interpola linealmente. Los esfuerzos se calculan como
$\mathbf{f}_e=\mathbf{k}_e\mathbf{u}_e$ en cada elemento: $M_f(z)$ es lineal y
$V(z)=-dM_f/dz$ es constante dentro de cada elemento, porque no hay carga
distribuida.

Las reacciones en el empotramiento son $R_z=1$ kN, $|M_y|=0.5$ kNm y
$|M_x|=2$ kNm para todas las mallas, y coinciden con el equilibrio
($P$, $P\,L/2$ y $M$).

\textbf{Discusión}
\begin{itemize}[itemsep=4pt]
    \item \textbf{Momento y corte exactos en todas las mallas.} El voladizo es
    isostático y las cargas actúan en los nodos, así que los esfuerzos
    internos $\mathbf{f}_e=\mathbf{k}_e\mathbf{u}_e$ quedan determinados por el
    equilibrio y no por la rigidez. Por eso $M_f(0)=P\,L/2=500$ N$\cdot$m,
    $V=P=1000$ N y $T=M=2000$ N$\cdot$m se obtienen exactos incluso con $n=2$.
    Además, como $M_f$ es lineal y $V$ y $T$ son constantes en cada tramo, la
    interpolación del elemento los representa sin error: el MEF coincide con
    la solución analítica en toda la viga y no solo en los nodos
    (Figura~\ref{fig:t2-p1-VMT}).

    \item \textbf{Flexión: convergencia de orden $l^4$ en los nodos.} Con
    $I(z)$ variable, la curvatura exacta $M_f/(EI)$ no es lineal dentro de un
    elemento y las cúbicas de Hermite no pueden reproducirla. Aun así, el
    error nodal es muy pequeño con $n=2$ ($0.11$\,\% en $v(L/2)$ y $0.03$\,\%
    en $v(L)$), y se divide por $\approx 16=2^4$ cada vez que $n$ se duplica
    (pendiente $4$ en el gráfico log-log), que es la superconvergencia nodal
    típica del elemento de Hermite. En los gráficos de $v$ y $\theta$ las
    curvas MEF son prácticamente indistinguibles de la analítica.

    \item \textbf{Torsión: convergencia de orden $l^2$.} El elemento lineal
    usa el promedio aritmético $\bar J$, que para $J$ lineal es mayor que el
    que da la rigidez exacta del tramo, $l\big/\!\int_0^l dz/J(z)$ (promedio
    armónico). Por eso el MEF es más rígido y \emph{subestima} el giro:
    $\varphi(L)$ tiene un error de $1.1$\,\% con $n=2$, que baja a $0.02$\,\%
    con $n=16$ y se divide por $\approx 4=2^2$ cada vez que se duplica $n$.

    \item \textbf{Tramo $L/2<z\le L$.} Como ahí $M_f=0$, ese tramo se mueve
    como sólido rígido: $\theta$ es constante y $v$ crece linealmente. El
    MEF lo reproduce exactamente, porque un polinomio cúbico contiene la
    recta. Todo el error de flexión del extremo viene del tramo
    $0\le z\le L/2$. La torsión, en cambio, actúa en toda la viga y
    acumula error en todos los elementos.

    \item \textbf{Elección de malla.} Para flexión bastaría con $n=2$ (error
    del orden de $0.1$\,\%). Para la torsión se necesita al menos $n=8$ para
    bajar del $0.1$\,\%. Con $n=16$ todos los campos quedan convergidos para
    fines de diseño.
\end{itemize}
